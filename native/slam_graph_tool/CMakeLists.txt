cmake_minimum_required(VERSION 3.16)
project(slam_graph_tool)
find_package(ament_cmake REQUIRED)
find_package(slam_toolbox REQUIRED)
find_package(Boost REQUIRED COMPONENTS serialization filesystem system thread)
find_package(Eigen3 REQUIRED)
get_filename_component(ROS_PREFIX "${slam_toolbox_DIR}/../../.." ABSOLUTE)
find_library(KARTO_LIBRARY kartoSlamToolbox HINTS "${ROS_PREFIX}/lib" REQUIRED)
add_executable(slam_graph_tool main.cpp)
target_include_directories(slam_graph_tool PRIVATE ${EIGEN3_INCLUDE_DIRS})
ament_target_dependencies(slam_graph_tool slam_toolbox)
target_link_libraries(slam_graph_tool ${KARTO_LIBRARY} ${Boost_LIBRARIES} tbb)

add_executable(slam_region_tool region.cpp)
target_include_directories(slam_region_tool PRIVATE ${EIGEN3_INCLUDE_DIRS})
ament_target_dependencies(slam_region_tool slam_toolbox)
target_link_libraries(slam_region_tool ${KARTO_LIBRARY} ${Boost_LIBRARIES} tbb)

add_executable(slam_region_audit audit.cpp)
target_include_directories(slam_region_audit PRIVATE ${EIGEN3_INCLUDE_DIRS})
ament_target_dependencies(slam_region_audit slam_toolbox)
target_link_libraries(slam_region_audit ${KARTO_LIBRARY} ${Boost_LIBRARIES} tbb)

include(CTest)
if(BUILD_TESTING)
  add_executable(test_padded_grid test_padded_grid.cpp)
  target_include_directories(test_padded_grid PRIVATE ${EIGEN3_INCLUDE_DIRS})
  ament_target_dependencies(test_padded_grid slam_toolbox)
  target_link_libraries(test_padded_grid ${KARTO_LIBRARY} ${Boost_LIBRARIES} tbb)
  add_test(NAME observed_boundary_endpoints COMMAND test_padded_grid)
  add_executable(test_region_pair test_region_pair.cpp)
  target_compile_features(test_region_pair PRIVATE cxx_std_17)
  target_include_directories(test_region_pair PRIVATE ${EIGEN3_INCLUDE_DIRS})
  ament_target_dependencies(test_region_pair slam_toolbox)
  target_link_libraries(test_region_pair ${KARTO_LIBRARY} ${Boost_LIBRARIES} tbb)
  add_test(NAME region_pair_persistence COMMAND test_region_pair
    $<TARGET_FILE:slam_region_tool> ${CMAKE_CURRENT_BINARY_DIR}/region-fixture)
endif()
